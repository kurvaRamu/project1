pipeline {
    agent any

    tools {
        maven 'Maven-3.9.16'
    }

    stages {
        stage('Checkout') {
            steps {
                git branch: 'main',
                    url: 'https://github.com/kurvaRamu/my_web_app.git'
            }
        }

        stage('Build') {
            steps {
                bat 'mvn clean package'
            }
        }

        stage('Test') {
            steps {
                bat 'mvn test'
            }
        }

        stage('Docker Build') {
            steps {
                bat 'docker build -t webimage .'
                bat 'docker tag webimage kurvaramu/webimage:v%BUILD_NUMBER%'
                bat 'docker tag webimage kurvaramu/webimage:latest'
            }
        }

        stage('Docker Login') {
            steps {
                withCredentials([
                    usernamePassword(
                        credentialsId: 'dockerhub-credentials',
                        usernameVariable: 'DOCKER_USERNAME',
                        passwordVariable: 'DOCKER_PASSWORD'
                    )
                ]) {
                    bat '''
                        @echo off
                        echo %DOCKER_PASSWORD% | docker login -u %DOCKER_USERNAME% --password-stdin
                    '''
                }
            }
        }

        stage('Docker Push') {
            steps {
                bat 'docker push kurvaramu/webimage:v%BUILD_NUMBER%'
                bat 'docker push kurvaramu/webimage:latest'
            }
        }

        stage('Docker Run') {
            steps {
                bat '''
                    @echo off
                    docker rm -f my-web-container 2>NUL
                    docker run -d --name my-web-container -p 8094:8080 kurvaramu/webimage:latest
                '''
            }
        }
    }

    post {
        success {
            mail to: 'k.ramu51797@gmail.com',
                 subject: "Success: Pipeline ${env.JOB_NAME} [Build #${env.BUILD_NUMBER}]",
                 body: "The pipeline completed successfully!\n\nView the logs here: ${env.BUILD_URL}"
        }

        failure {
            mail to: 'k.ramu51797@gmail.com',
                 subject: "Failure: Pipeline ${env.JOB_NAME} [Build #${env.BUILD_NUMBER}]",
                 body: "The pipeline has failed.\n\nCheck the console output: ${env.BUILD_URL}console"
        }
    }
}
